#encryption and decryption
touch plaintext.txt 
open plaintext.txt 
openssl enc -aes256 -salt -in plaintext.txt -out encrypted.enc 
Passphrase:12345678

openssl enc -d -aes256 -in encrypted.enc -out decrypted.txt
