contract A 
{

{

}

B.sol 
// SPDX-License-Identifier: MIT 
pragma solidity ^0.8.0; 
import "./A.sol"; 
contract B 
{ 
A a1; 
constructor(address _a1) 
{ 
a1 = A(_a1); 
} 
function showB()public view returns(string memory) 
{ 
return a1.showA(); 
}  }
